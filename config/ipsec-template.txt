! Educational Packet Tracer template. Check IOS/device support before use.
! Substitute a unique disposable lab key locally; never commit real secrets.
crypto isakmp policy 10
 encryption aes 256
 hash sha256
 authentication pre-share
 group 14
 lifetime 28800
!
crypto isakmp key <LOCAL_LAB_ONLY_SECRET> address <REMOTE_WAN_PEER>
!
crypto ipsec transform-set CWN-SET esp-aes 256 esp-sha-hmac
 mode tunnel
!
ip access-list extended VPN-TRAFFIC
 permit ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
!
crypto map CWN-MAP 10 ipsec-isakmp
 set peer <REMOTE_WAN_PEER>
 set transform-set CWN-SET
 match address VPN-TRAFFIC
!
interface <WAN_INTERFACE>
 crypto map CWN-MAP
!
